function toCSV() {

function doImport(file) {
  const fr = new FileReader();
  fr.onload = () => {
    try {
      const parsed = JSON.parse(String(fr.result));
      const incoming = Array.isArray(parsed) ? parsed : parsed.rows;
      if (!Array.isArray(incoming)) throw new Error("no rows");
      const clean = incoming.filter((r) => r && typeof r === "object").map((r) => {
        const copy = Object.assign({}, r);
        if (!copy.id) copy.id = uid();
        return copy;

      banner = {kind: "bad", text: "That file is not a valid backup: " + e.message};
      paintBanners();
    }

}
